1. Get the application URL by running these commands:
{{- if contains "NodePort" .Values.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Values.namespace | default .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "webapp.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Values.namespace | default .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "ClusterIP" .Values.service.type }}
  export POD_NAME=$(kubectl get pods --namespace {{ .Values.namespace | default .Release.Namespace }} -l "app.kubernetes.io/name={{ include "webapp.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}}")
  export CONTAINER_PORT=$(kubectl get pod --namespace {{ .Values.namespace | default .Release.Namespace }} $POD_NAME -o jsonpath="{.spec.containers[0].ports[0].containerPort}}")
  echo "Visit http://127.0.0.1:8080 to use your application"
  kubectl port-forward --namespace {{ .Values.namespace | default .Release.Namespace }} $POD_NAME 8080:$CONTAINER_PORT
{{- end }}

2. The API key is stored in a Secret and mounted at /etc/secrets/api-key
3. Config files from ConfigMap are mounted at /etc/nginx/conf.d
